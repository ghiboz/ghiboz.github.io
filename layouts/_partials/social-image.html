{{- /* Returns the image used in social cards (Open Graph and X/Twitter) for the page in context: a dict with url, alt, width and height. - If the article has `cover.image` and the file is in its folder (page bundle), Hugo makes a 1200x630 JPEG from it. X wants a JPG/PNG/WebP/GIF under 5 MB, and Unsplash originals are often far bigger, so the original is never used for the card. - If the cover is an external address, that address is used as it is. - Without a cover, the site default (params.defaultSocialImage) is used. */ -}} {{- $url := "" }} {{- $alt := "" }} {{- $width := 0 }} {{- $height := 0 }} {{- with .Params.cover }} {{- with .image }} {{- $alt = $.Params.cover.alt | default "" | plainify }} {{- $img := ($.Resources.ByType "image").GetMatch (printf "*%s*" .) }} {{- if not $img }} {{- $img = (resources.ByType "image").GetMatch (printf "*%s*" .) }} {{- end }} {{- if and $img (in (slice "jpg" "jpeg" "png" "webp") $img.MediaType.SubType) }} {{- $card := $img.Fill "1200x630 q85 jpg" }} {{- $url = $card.Permalink }} {{- $width = $card.Width }} {{- $height = $card.Height }} {{- else if or (hasPrefix . "http") (hasPrefix . "/") }} {{- $url = . | absURL }} {{- else }} {{- warnf "cover image %q not found for %s: no card image" . $.File.Path }} {{- end }} {{- end }} {{- end }} {{- if not $url }} {{- with site.Params.defaultSocialImage }} {{- $url = . | absURL }} {{- $width = 1200 }} {{- $height = 630 }} {{- $alt = site.Title }} {{- end }} {{- end }} {{- return (dict "url" $url "alt" $alt "width" $width "height" $height) }}
